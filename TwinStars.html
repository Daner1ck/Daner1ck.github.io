               gain.gain.exponentialRampToValueAtTime(0.01, t + 0.15);
               osc.start(t);
               osc.stop(t + 0.15);
           }

               osc.type = 'triangle';

               gain.gain.setValueAtTime(0.2, t);
               gain.gain.linearRampToValueAtTime(0, t + 0.1);
               osc.start(t);
               osc.stop(t + 0.1);

       const TILE = 40;
       const GRAVITY = 0.5;
       const FRICTION = 0.85;
       const SPEED = 0.6;
       const MAX_SPEED = 4;
       const JUMP_POWER = -9;

       let currentLevelIndex = 0;
       let map = [];
       let activeId = 'pink';
       let gameState = 'start'; // start, playing, transition
       let keys = { left: false, right: false, up: false };

       class Player {
           constructor(id, color, startCol, startRow) {
               this.id = id;
               this.color = color;
               this.w = 28; // Ligeramente más delgado para no atascarse
               this.h = 28;
               // Centrar en el bloque
               this.x = startCol * TILE + (TILE - this.w) / 2;
               this.y = startRow * TILE + (TILE - this.h);
               this.vx = 0;
               this.vy = 0;
               this.grounded = false;
               this.facingRight = true;
           }

           update(otherPlayer) {
               if (gameState !== 'playing') return;
               const isActive = (this.id === activeId);

               // Inputs solo si está activo
               if (isActive) {
                   if (keys.left) { this.vx -= SPEED; this.facingRight = false; }
                   if (keys.right) { this.vx += SPEED; this.facingRight = true; }
                   if (keys.up && this.grounded) {
                       this.vy = JUMP_POWER;
                       this.grounded = false;
                       playSound('jump');
                       keys.up = false; // Evita salto automático continuo
                   }
               }

               // Físicas básicas
               this.vx *= FRICTION;
               this.vy += GRAVITY;

               // Limitar velocidad
               if (this.vx > MAX_SPEED) this.vx = MAX_SPEED;
               if (this.vx < -MAX_SPEED) this.vx = -MAX_SPEED;
               if (Math.abs(this.vx) < 0.1) this.vx = 0;
               if (this.vy > 12) this.vy = 12; // Terminal velocity

               // Mover en X y comprobar colisiones
               this.x += this.vx;
               this.checkMapCollisions('x');
               this.checkPlayerCollision(otherPlayer, 'x');

               // Mover en Y y comprobar colisiones
               this.y += this.vy;
               this.grounded = false;
               this.checkMapCollisions('y');
               this.checkPlayerCollision(otherPlayer, 'y');

               // Limitar con los bordes de la pantalla
               if (this.x < 0) this.x = 0;
               if (this.x + this.w > canvas.width) this.x = canvas.width - this.w;
               if (this.y > canvas.height) {
                   // Caída al vacío, reiniciar nivel automáticamente
                   initLevel(currentLevelIndex);
               }
           }

           isSolid(tile) {
               if (tile === 1) return true; // Gris sólido para ambos
               if (tile === 2 && this.id === 'pink') return true; // Rosa sólido para Rosa
               if (tile === 3 && this.id === 'blue') return true; // Azul sólido para Azul
               return false;
           }

           checkMapCollisions(axis) {
               const startCol = Math.floor(this.x / TILE);
               const endCol = Math.floor((this.x + this.w) / TILE);
               const startRow = Math.floor(this.y / TILE);
               const endRow = Math.floor((this.y + this.h) / TILE);

               for (let r = startRow; r <= endRow; r++) {
                   for (let c = startCol; c <= endCol; c++) {
                       if (r >= 0 && r < map.length && c >= 0 && c < map[0].length) {
                           if (this.isSolid(map[r][c])) {
                               this.resolveCollision(c * TILE, r * TILE, TILE, TILE, axis);
                           }
                       }
                   }
               }
           }

           checkPlayerCollision(other, axis) {
               // AABB Check
               if (this.x < other.x + other.w &&
                   this.x + this.w > other.x &&
                   this.y < other.y + other.h &&
                   this.y + this.h > other.y) {
                   
                   this.resolveCollision(other.x, other.y, other.w, other.h, axis);
                   
                   // Si el activo cae sobre el inactivo, se considera grounded
                   if (axis === 'y' && this.y < other.y && this.id === activeId) {
                       this.grounded = true;
                   }
               }
           }

           resolveCollision(tx, ty, tw, th, axis) {
               if (axis === 'x') {
                   if (this.vx > 0) { // Hacia la derecha
                       this.x = tx - this.w;
                   } else if (this.vx < 0) { // Hacia la izquierda
                       this.x = tx + tw;
                   }
                   this.vx = 0;
               } else if (axis === 'y') {
                   if (this.vy > 0) { // Cayendo
                       this.y = ty - this.h;
                       this.grounded = true;
                   } else if (this.vy < 0) { // Chocando techo
                       this.y = ty + th;
                   }
                   this.vy = 0;
               }
           }

           draw() {
               const isActive = (this.id === activeId);
               
               ctx.save();
               ctx.translate(this.x, this.y);

               // Transparencia si es inactivo
               if (!isActive) ctx.globalAlpha = 0.6;

               // Cuerpo
               ctx.fillStyle = this.color;
               ctx.beginPath();
               ctx.roundRect(0, 0, this.w, this.h, 8);
               ctx.fill();

               // Cara y Ojos
               ctx.fillStyle = 'white';
               const eyeY = 8;
               const eyeX = this.facingRight ? 4 : -4;
               const centerX = this.w / 2;

               if (isActive) {
                   // Ojos abiertos
                   ctx.beginPath();
                   ctx.arc(centerX - 5 + eyeX, eyeY, 3, 0, Math.PI * 2);
                   ctx.arc(centerX + 5 + eyeX, eyeY, 3, 0, Math.PI * 2);
                   ctx.fill();
                   
                   // Pupilas
                   ctx.fillStyle = '#333';
                   ctx.beginPath();
                   ctx.arc(centerX - 4 + eyeX, eyeY, 1.5, 0, Math.PI * 2);
                   ctx.arc(centerX + 6 + eyeX, eyeY, 1.5, 0, Math.PI * 2);
                   ctx.fill();

                   // Boquita feliz
                   ctx.strokeStyle = '#333';
                   ctx.lineWidth = 1.5;
                   ctx.beginPath();
                   ctx.arc(centerX + eyeX, eyeY + 6, 3, 0, Math.PI);
                   ctx.stroke();
               } else {
                   // Ojos durmiendo
                   ctx.lineWidth = 2;
                   ctx.strokeStyle = 'white';
                   ctx.beginPath();
                   ctx.moveTo(centerX - 7 + eyeX, eyeY);
                   ctx.lineTo(centerX - 3 + eyeX, eyeY);
                   ctx.moveTo(centerX + 3 + eyeX, eyeY);
                   ctx.lineTo(centerX + 7 + eyeX, eyeY);
                   ctx.stroke();
               }

               ctx.restore();
           }
       }

       let p1, p2;

       function initLevel(index) {
           if (index >= levelsData.length) {
               gameState = 'end';
               screens.end.classList.add('active');
               return;
           }

           const rawMap = levelsData[index];
           map = [];
           
           let pSpawn = {c: 0, r: 0};
           let bSpawn = {c: 0, r: 0};

           for (let r = 0; r < rawMap.length; r++) {
               let row = [];
               for (let c = 0; c < rawMap[r].length; c++) {
                   let char = rawMap[r][c];
                   if (char === 'p') { pSpawn = {c, r}; row.push(0); }
                   else if (char === 'b') { bSpawn = {c, r}; row.push(0); }
                   else { row.push(parseInt(char)); }
               }
               map.push(row);
           }

           p1 = new Player('pink', 'var(--pink-main)', pSpawn.c, pSpawn.r);
           p2 = new Player('blue', 'var(--blue-main)', bSpawn.c, bSpawn.r);
           
           activeId = 'pink';
           swapBtn.classList.remove('blue-turn');
           levelDisplay.innerText = `Nivel ${index + 1}`;
           gameState = 'playing';
           
           // Limpiar inputs
           keys.left = keys.right = keys.up = false;
       }

       function drawMap() {
           for (let r = 0; r < map.length; r++) {
               for (let c = 0; c < map[0].length; c++) {
                   let tile = map[r][c];
                   if (tile === 0) continue;

                   let x = c * TILE;
                   let y = r * TILE;

                   ctx.beginPath();
                   ctx.roundRect(x + 1, y + 1, TILE - 2, TILE - 2, 6);

                   if (tile === 1) ctx.fillStyle = 'var(--neutral)';
                   else if (tile === 2) ctx.fillStyle = 'var(--pink-light)';
                   else if (tile === 3) ctx.fillStyle = 'var(--blue-light)';
                   else if (tile === 4) {
                       // Estrella/Meta
                       ctx.fillStyle = 'var(--gold)';
                       ctx.shadowColor = 'var(--gold)';
                       ctx.shadowBlur = 10;
                   }
                   
                   ctx.fill();
                   ctx.shadowBlur = 0;
               }
           }
       }

       function checkWin() {
           if (gameState !== 'playing') return;

           const inGoal = (p) => {
               // Comprobar el centro del jugador
               let c = Math.floor((p.x + p.w / 2) / TILE);
               let r = Math.floor((p.y + p.h / 2) / TILE);
               return (map[r] && map[r][c] === 4);
           };

           if (inGoal(p1) && inGoal(p2)) {
               gameState = 'transition';
               playSound('win');
               setTimeout(() => {
                   screens.win.classList.add('active');
               }, 400);
           }
       }

       function togglePlayer() {
           if (gameState !== 'playing') return;
           activeId = activeId === 'pink' ? 'blue' : 'pink';
           swapBtn.classList.toggle('blue-turn', activeId === 'blue');
           playSound('swap');
           // Liberar inputs al cambiar para evitar bloqueos
           keys.left = keys.right = keys.up = false;
       }

       function gameLoop() {
           ctx.clearRect(0, 0, canvas.width, canvas.height);

           if (gameState !== 'start') {
               if (activeId === 'pink') {
                   p1.update(p2);
                   p2.update(p1);
               } else {
                   p2.update(p1);
                   p1.update(p2);
               }

               checkWin();
               drawMap();
               
               // Dibujar primero al inactivo, luego al activo
               if (activeId === 'pink') { p2.draw(); p1.draw(); }
               else { p1.draw(); p2.draw(); }
           }

           requestAnimationFrame(gameLoop);
       }

       window.addEventListener('keydown', (e) => {
           initAudio();

           if (e.code === 'ArrowRight') keys.right = true;
           if (e.code === 'ArrowUp') keys.up = true;
           if (e.code === 'KeyC') togglePlayer();
       });

       window.addEventListener('keyup', (e) => {
           if (e.code === 'ArrowLeft') keys.left = false;
           if (e.code === 'ArrowRight') keys.right = false;
           if (e.code === 'ArrowUp') keys.up = false;
       });

       // Configuración Botones en Pantalla
       function setupBtn(id, onDown, onUp) {
           const btn = document.getElementById(id);
           const press = (e) => { e.preventDefault(); initAudio(); btn.classList.add('pressed'); onDown(); };
           const release = (e) => { e.preventDefault(); btn.classList.remove('pressed'); onUp(); };
           
           btn.addEventListener('touchstart', press, { passive: false });
           btn.addEventListener('touchend', release, { passive: false });
           btn.addEventListener('mousedown', press);
           btn.addEventListener('mouseup', release);
           btn.addEventListener('mouseleave', release);
       }

       setupBtn('btn-left', () => keys.left = true, () => keys.left = false);
       setupBtn('btn-right', () => keys.right = true, () => keys.right = false);
       setupBtn('btn-jump', () => keys.up = true, () => keys.up = false);
       setupBtn('btn-swap', () => togglePlayer(), () => {});
       
       document.getElementById('btn-restart-small').addEventListener('click', () => {
           initAudio();
           if (gameState === 'playing') initLevel(currentLevelIndex);
       });

           initAudio();
           screens.start.classList.remove('active');
           currentLevelIndex = 0;
           initLevel(0);
       });

           screens.win.classList.remove('active');
           currentLevelIndex++;
           initLevel(currentLevelIndex);
       });
       
       document.getElementById('btn-restart-all').addEventListener('click', () => {
           screens.end.classList.remove('active');
           currentLevelIndex = 0;
           initLevel(0);
       });

       // Iniciar bucle de renderizado en reposo
       requestAnimationFrame(gameLoop);

   </script>
